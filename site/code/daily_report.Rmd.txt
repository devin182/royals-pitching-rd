```{sql, connection=con, output.var="lines"}
-- each Royals pitcher's line vs. his own season-to-date baseline (all games before today)
WITH today AS (
  SELECT pitcher, any_value(player_name) AS name, pitch_type, count(*) n,
         avg(release_speed) velo, max(release_speed) vmax, avg(ivb) ivb, avg(hb_arm) hb, avg(release_spin_rate) spin,
         avg(release_pos_z) rz, avg(rel_side_arm) rx, avg(release_extension) ext,
         sum(is_whiff::INT) whiffs, sum(is_swing::INT) swings, sum((is_whiff OR is_called_strike)::INT) csw
  FROM pitch_features WHERE is_royals AND game_date = ?day_s GROUP BY ALL),
base AS (
  SELECT pitcher, pitch_type, count(*) bn, avg(release_speed) velo, avg(ivb) ivb, avg(hb_arm) hb,
         avg(release_spin_rate) spin, avg(release_pos_z) rz, avg(rel_side_arm) rx, avg(release_extension) ext
  FROM pitch_features WHERE is_royals AND game_date < ?day_s GROUP BY ALL)
SELECT t.*, b.bn, t.velo - b.velo dv, t.ivb - b.ivb divb, t.hb - b.hb dhb, t.spin - b.spin dspin,
       12 * sqrt(power(t.rz - b.rz, 2) + power(t.rx - b.rx, 2)) drel, t.ext - b.ext dext
FROM today t LEFT JOIN base b USING (pitcher, pitch_type)
ORDER BY name, n DESC
```

```{sql, connection=con, output.var="work"}
-- workload: pitches today, days of rest, pitches over the last 7 days
WITH d AS (SELECT pitcher, any_value(player_name) AS name, game_date, count(*) n
           FROM pitch_features WHERE is_royals AND game_date <= ?day_s GROUP BY pitcher, game_date)
SELECT name, max(n) FILTER (WHERE game_date = ?day_s) today,
       date_diff('day', max(game_date) FILTER (WHERE game_date < ?day_s), CAST(?day_s AS DATE)) - 1 AS rest_before,
       sum(n) FILTER (WHERE game_date > CAST(?day_s AS DATE) - INTERVAL 7 DAY) last7,
       count(*) FILTER (WHERE game_date > CAST(?day_s AS DATE) - INTERVAL 3 DAY) apps_3d
FROM d GROUP BY name HAVING max(n) FILTER (WHERE game_date = ?day_s) IS NOT NULL ORDER BY today DESC
```

```{sql, connection=con, output.var="hit"}
-- Opponent hitters who batted today: season results vs. each pitch group (all MLB pitching), with shrinkage.
WITH lineup AS (SELECT DISTINCT batter FROM pitch_features WHERE is_royals AND game_date = ?day_s),
s AS (SELECT batter, pitch_group, count(*) n, sum(xrv) xrv, sum(is_whiff::INT) wh, sum(is_swing::INT) sw,
             sum(is_chase::INT) ch, sum((NOT in_zone)::INT) oz,
             avg(xwoba_con) FILTER (WHERE type = 'X') xwcon
      FROM pitch_features WHERE batter IN (SELECT batter FROM lineup) AND game_date <= ?day_s GROUP BY ALL),
lg AS (SELECT pitch_group, avg(xrv) lxrv, avg(is_whiff::INT) FILTER (WHERE is_swing) lwh,
              avg(is_chase::INT) FILTER (WHERE NOT in_zone) lch FROM pitch_features GROUP BY 1)
SELECT coalesce(p.full_name, s.batter::VARCHAR) hitter, p.bats, s.*, lg.lxrv, lg.lwh, lg.lch
FROM s JOIN lg USING (pitch_group) LEFT JOIN players p ON p.id = s.batter
```

```{r}
hit <- as.data.table(hit)
# hitter perspective: pitcher xRV is + when the pitch wins, so hitter value = -xRV. Shrink toward league.
hit[, `:=`(rv100 = 100 * (xrv + K * lxrv) / (n + K), whiff = (wh + K / 3 * lwh) / (sw + K / 3), chase = (ch + K / 2 * lch) / (oz + K / 2))]
w <- dcast(hit, hitter + bats ~ pitch_group, value.var = c("n", "rv100", "whiff", "chase"))
plan <- hit[, {
  o <- order(-rv100); best <- pitch_group[o[1]]; worst <- pitch_group[o[.N]]
  pa <- pitch_group[which.max(whiff)]
  .(attack = sprintf("%s (%+.1f)", best, max(rv100)), avoid = sprintf("%s (%+.1f)", worst, min(rv100)),
    putaway = sprintf("%s (%s whiff)", pa, pct(max(whiff))), seen = sum(n), chase_all = sum(ch) / sum(oz))
}, by = .(hitter, bats)][order(-seen)]
```
